%% DGADashboard_code.txt - the hand-written code of DGADashboard.mlapp (roadmap Steps 39 and 40, decisions D37 and D38; master control 15 Oct 2026; redesign 17 Oct 2026).
%% build_dashboard_mlapp.m reads this file and puts every section into the app. Section lines start with "%%@".
%% Callbacks follow MATLAB_Build_Guide Part C (C4.1 to C4.5); additions are marked (D37), (D38), (MC) = master control 15 Oct 2026, (RD) = redesign 17 Oct 2026.
%%@ PUBLIC_PROPERTIES
    properties (Access = public)
        DGAFile        % full path of the measured DGA file
        ServiceURL     % address of the FastAPI service (D37)
        LastResult     % struct of the last diagnosis: record_id, gas, conventional, ml, agreement, verdict (D37, RD)
        SimResult      % struct returned by run_transformer_simulation (public so that the checks can compare it) (MC)
        SimEvents      % table of the sampling events of the last simulation: time_h, record_index, record_id (D38)
        EventResults   % struct array, one element per sampling event diagnosed in the last run (D38)
        PlaybackPause = 0.5  % pause between sampling events in the playback, s (0 in tests) (D38)
        ExplainState   % struct with the explanation texts: sim, diag, events (string columns), focus ("diag" or "run") (MC)
        ProjRoot       % folder that holds project_setup.m (MC)
        ServiceStarted = false  % true when THIS app started the ML service; only then it stops it when it closes (MC)
        SampleSeed = 1  % seed of the next random choice of records; it grows by 1 after every random choice (RD)
        Chosen         % struct rows / record_id / label of the samples of the current run (RD)
    end
%%@ PRIVATE_PROPERTIES
    properties (Access = private)
        Gas        % struct with H2, CH4, C2H6, C2H4, C2H2 of the current record
        Info        % table row of the current record (traceability)
    end
%%@ PRIVATE_METHODS
    methods (Access = private)

        % Lamp and text of the agreement flag (Build Guide C4.5; rule in dga_agreement.m, D37)
        function updateAgreement(app, a)
            app.AgreementLamp.Color = a.color;
            app.AgreementLabel.Text = a.label;
        end

        % Text of the status line and of a progress dialog while a long step runs (MC)
        function setProgress(app, dlg, txt)
            app.StatusLabel.Text = txt;
            if ~isempty(dlg) && isvalid(dlg), dlg.Message = txt; end
            drawnow limitrate;
        end

        % Switch the buttons off while a long step runs, so that a second click cannot start a second step (MC, RD)
        function setBusy(app, tf)
            if ~isvalid(app), return, end
            st = 'on'; if tf, st = 'off'; end
            app.RunSimulationButton.Enable = st;  app.LoadDGAFileButton.Enable = st;  app.DiagnoseButton.Enable = st;
            app.RestartServiceButton.Enable = st;  app.RecordSpinner.Enable = st;
            app.DiagnoseValuesButton.Enable = st;  app.ScenarioDropDown.Enable = st;  app.SampleSetDropDown.Enable = st;  app.FaultFilterDropDown.Enable = st;  app.RandomRecordButton.Enable = st;
            drawnow;
        end

        % Show the explanation text of the current state in the right-hand panel (MC)
        function refreshExplanation(app)
            app.ExplainArea.Value = cellstr(dashboard_explain('all', app.ExplainState));
            try, scroll(app.ExplainArea, 'top'); catch, end
        end

        % Is the local ML service answering? If not, (re)start it before the call. A service that hangs is stopped and started again (MC)
        function ok = mlReady(app)
            ok = true;
            if ~contains(app.ServiceURL, '127.0.0.1:8000'), return, end    % another address: nothing to manage
            try
                r = webread('http://127.0.0.1:8000/health', weboptions('Timeout', 2));
                ok = isstruct(r);
            catch
                ok = false;
            end
            if ~ok
                [ok, ~, started] = ensure_ml_service(app.ProjRoot, 60, @(s) app.setProgress([], s));
                if started, app.ServiceStarted = true; end
            end
        end

        % The chart areas are always drawn on white with dark text, so that they read well in the light and in the dark theme of MATLAB (RD)
        function styleAxes(~, ax)
            ink = [0.15 0.15 0.15];
            ax.Color = [1 1 1];  ax.XColor = ink;  ax.YColor = ink;  ax.GridColor = [0.5 0.5 0.5];
            try, ax.Title.Color = ink; ax.XLabel.Color = ink; ax.YLabel.Color = ink; catch, end
        end

        % Colour of a banner or card: panel and its grid (a grid does not take the colour of its parent) (RD)
        function tint(app, comps, rgb)
            for c = comps(:)'
                c{1}.BackgroundColor = rgb;
            end
        end

        % Start screen of the verdict banner, the cards and the charts: nothing diagnosed yet (RD)
        function showIdle(app)
            grey = [0.45 0.45 0.45];
            app.tint({app.VerdictPanel, app.VerdictGrid, app.AgreementGrid}, grey);
            app.VerdictLabel.Text = 'WAITING FOR A SAMPLE';
            app.VerdictDetail.Text = 'Choose the operating condition and the samples on the left, then press "Run simulation + diagnose". Or pick one record with "Pick random record".';
            app.VerdictAction.Text = 'The answer appears here: HEALTHY (green), FAULT with its type (red), or WATCH / CHECK (amber).';
            cards = {'Rogers', 'IEC', 'Duval', 'ML', 'Ref'};
            for k = 1:numel(cards)
                app.([cards{k} 'Answer']).Text = '-';  app.([cards{k} 'Note']).Text = '';  app.([cards{k} 'Judge']).Text = '';
                app.tint({app.([cards{k} 'Card']), app.([cards{k} 'Grid'])}, [0.94 0.94 0.94]);
            end
            cla(app.GasAxes);  cla(app.ProbAxes);  cla(app.UIAxesDuval);
            title(app.GasAxes, 'Gas level compared with the typical value');  title(app.ProbAxes, 'ML probability of each fault type');
            app.styleAxes(app.GasAxes);  app.styleAxes(app.ProbAxes);  app.styleAxes(app.UIAxesTemperature);
            app.StepsArea.Value = {'The standard procedure for the sample on the screen appears here after a diagnosis.'};
        end

        % Banner: the answer in one line, why, and what to do (RD)
        function showVerdict(app, V)
            app.tint({app.VerdictPanel, app.VerdictGrid, app.AgreementGrid}, V.color);
            app.VerdictLabel.Text = char(V.title);
            app.VerdictDetail.Text = char(V.detail);
            app.VerdictAction.Text = char("What to do: " + V.action);
        end

        % The four method cards and the card of the known label: answer, plain meaning, AGREES / DIFFERS (RD)
        function showCards(app, V)
            cards = {'Rogers', 'IEC', 'Duval', 'ML'};
            for k = 1:4
                x = V.answers(k);  if x == "", x = "n/a"; end
                note = dashboard_explain('label', V.answers(k));
                if V.has_ref && V.answers(k) ~= ""
                    if V.ref_ok(k), note = note + newline + "Known label: right"; else, note = note + newline + "Known label: wrong"; end
                end
                app.([cards{k} 'Answer']).Text = char(x);
                app.([cards{k} 'Note']).Text = char(note);
                app.([cards{k} 'Judge']).Text = char(V.judge(k));
                switch V.judge(k)
                    case "AGREES",  col = [0.86 0.95 0.86];  fc = [0 0.4 0];
                    case "DIFFERS", col = [0.98 0.87 0.87];  fc = [0.7 0 0];
                    otherwise,      col = [0.93 0.93 0.93];  fc = [0.35 0.35 0.35];
                end
                app.tint({app.([cards{k} 'Card']), app.([cards{k} 'Grid'])}, col);
                app.([cards{k} 'Judge']).FontColor = fc;
            end
            if V.has_ref
                app.RefAnswer.Text = char(V.ref);
                app.RefNote.Text = char(dashboard_explain('label', V.ref));
                app.RefJudge.Text = 'from the data file';  app.RefJudge.FontColor = [0.1 0.1 0.1];
                app.tint({app.RefCard, app.RefGrid}, [0.88 0.92 0.98]);
            else
                app.RefAnswer.Text = '-';  app.RefNote.Text = 'This file has no known label for the record.';  app.RefJudge.Text = '';
                app.tint({app.RefCard, app.RefGrid}, [0.94 0.94 0.94]);
            end
        end

        % Bar chart: each gas as a multiple of its typical value (IEC 60599 Annex A); a bar above the dashed line is above typical (RD)
        function drawGas(app, V)
            ax = app.GasAxes;  cla(ax);
            G = V.gas;  y = max(G.times_typical, 0.01);
            b = bar(ax, 1:height(G), y, 'FaceColor', 'flat', 'BaseValue', 0.01);
            cd = repmat([0.2 0.6 0.3], height(G), 1);  cd(G.above, :) = repmat([0.8 0.2 0.2], sum(G.above), 1);
            b.CData = cd;
            ax.YScale = 'log';  ax.XTick = 1:height(G);  ax.XTickLabel = cellstr(G.gas);
            ylim(ax, [0.01 max(10, 5 * max(y))]);
            hold(ax, 'on');
            yline(ax, 1, 'k--', 'typical value', 'LabelHorizontalAlignment', 'right');
            for i = 1:height(G)
                text(ax, i, y(i), sprintf('%.8g ppm', G.ppm(i)), 'HorizontalAlignment', 'center', 'VerticalAlignment', 'bottom', 'FontSize', 9);
            end
            hold(ax, 'off');
            ylabel(ax, 'times the typical value');
            title(ax, 'Gas level (red = above typical value)');
            app.styleAxes(ax);
            for tx = findobj(ax, 'Type', 'text')', tx.Color = [0.1 0.1 0.1]; end
        end

        % Bar chart: probability of each fault type according to the ML model (RD)
        function drawProb(app, m)
            ax = app.ProbAxes;  cla(ax);
            if m.status ~= "OK"
                text(ax, 0.5, 0.5, {'ML not available'; char(m.status)}, 'HorizontalAlignment', 'center', 'Units', 'normalized');
                ax.XTick = [];  ax.YTick = [];  title(ax, 'ML probability of each fault type');  app.styleAxes(ax);
                for tx = findobj(ax, 'Type', 'text')', tx.Color = [0.1 0.1 0.1]; end
                return
            end
            order = ["Normal" "PD" "D1" "D2" "T1" "T2" "T3"];
            p = zeros(1, 7);
            for i = 1:7, j = find(string(m.classes) == order(i), 1); if ~isempty(j), p(i) = m.probabilities(j); end, end
            b = barh(ax, 1:7, 100 * p, 'FaceColor', 'flat');
            cd = repmat([0.55 0.65 0.8], 7, 1);  [~, im] = max(p);  cd(im, :) = [0.1 0.3 0.7];
            b.CData = cd;
            ax.YTick = 1:7;  ax.YTickLabel = cellstr(order);  ax.YDir = 'reverse';  xlim(ax, [0 100]);
            xlabel(ax, 'probability (%)');
            title(ax, 'ML probability of each fault type');
            app.styleAxes(ax);
        end

        % Show one finished diagnosis on the screen: verdict banner, method cards, charts, tables, steps and the explanation text (MC, RD)
        function showResult(app, R, row)
            g = R.gas;  d = R.conventional;  m = R.ml;  a = R.agreement;
            app.GasTable.Data = table(["H2"; "CH4"; "C2H6"; "C2H4"; "C2H2"], ...
                [g.H2; g.CH4; g.C2H6; g.C2H4; g.C2H2], 'VariableNames', {'Gas', 'ppm'});
            mlText = m.status;
            if m.status == "OK", mlText = m.fault_class; end
            yesno = repmat("-", 3, 1);
            if m.status == "OK", yesno(a.agrees) = "yes"; yesno(~a.agrees) = "no"; end
            app.ResultsTable.Data = table(["Rogers Ratio"; "IEC 60599 ratio"; "Duval Triangle 1"; "Machine learning"], ...
                [string(d.rogers); string(d.iec); string(d.duval); mlText], [yesno; "-"], ...
                'VariableNames', {'Method', 'Diagnosis', 'Agrees with ML'});
            if m.status == "OK"
                app.ProbabilityTable.Data = table(m.classes(:), m.probabilities(:), 'VariableNames', {'Class', 'Probability'});
                app.ModelLabel.Text = sprintf('ML: %s v%s (hash %s), confidence %.3f, %.0f ms', ...
                    m.model_name, m.model_version, extractBefore(m.model_sha256, 9), m.confidence, 1000 * m.latency_s);
            else
                app.ProbabilityTable.Data = table(strings(0, 1), zeros(0, 1), 'VariableNames', {'Class', 'Probability'});
                app.ModelLabel.Text = ['ML: ' char(m.status)];
            end
            try
                plot_duval_triangle(app.UIAxesDuval, d.ratios, fullfile(app.ProjRoot, '02_Standards', 'duval_triangle_zones.csv'));
                if d.gate_upper(1)
                    title(app.UIAxesDuval, 'Duval Triangle 1 (not used: gas level normal)');
                else
                    title(app.UIAxesDuval, 'Duval Triangle 1 (this sample)');
                end
                app.styleAxes(app.UIAxesDuval);
            catch ME
                cla(app.UIAxesDuval);
                app.StatusLabel.Text = ['Duval Triangle not drawn: ' ME.message];
            end
            app.updateAgreement(a);
            V = R.verdict;
            app.showVerdict(V);
            app.showCards(V);
            app.drawGas(V);
            app.drawProb(m);
            app.StepsArea.Value = cellstr(dashboard_explain('steps', g, d, m, a, V));
            app.ExplainState.diag = dashboard_explain('diagnosis', g, d, m, a);
            app.refreshExplanation();
        end

        % Diagnose one row of the loaded DGA file: the one code path of the Diagnose button, the random picker and the sampling events (D38, RD)
        function ok = diagnoseRow(app, row)
            ok = false;
            try
                [app.Gas, app.Info] = load_dga_record(app.DGAFile, row);
            catch ME
                app.StatusLabel.Text = ['Cannot read the record: ' ME.message];
                return
            end
            ref = "";                                      % known label of the record, only if the file has one (used for the right/wrong note, never for a diagnosis)
            if ismember('class_name', app.Info.Properties.VariableNames), ref = string(app.Info.class_name(1)); end
            ok = app.diagnoseGas(app.Gas, ref, row);
        end

        % Diagnose one set of five gas values (from a file row or typed in the table): conventional methods, ML, agreement, verdict, screen (RD)
        function ok = diagnoseGas(app, g, ref, row)
            ok = false;
            app.StatusLabel.Text = 'Diagnosing...'; drawnow;
            d = conventional_diagnosis(g);                 % three conventional diagnoses, computed here in MATLAB
            app.mlReady();                                 % the service answers, or it is restarted (MC)
            m = ml_diagnose(g, app.ServiceURL);            % ML diagnosis from the FastAPI service (webwrite)
            if m.status == "NOT_CONNECTED" && contains(app.ServiceURL, ':8000/')   % stopped or hung during the call: restart once and retry
                app.StatusLabel.Text = 'ML service not reachable: starting it...'; drawnow;
                [okS, ~, started] = ensure_ml_service(app.ProjRoot, 60, @(s) app.setProgress([], s));
                if started, app.ServiceStarted = true; end
                if okS, m = ml_diagnose(g, app.ServiceURL); end
            end
            a = dga_agreement(d, m);
            rid = ""; if isfield(g, 'record_id'), rid = g.record_id; end
            V = dashboard_verdict(g, d, m, a, ref);
            app.LastResult = struct('record_id', rid, 'gas', g, 'conventional', d, 'ml', m, 'agreement', a, 'verdict', V);
            app.ExplainState.focus = "diag";
            app.showResult(app.LastResult, row);
            if startsWith(app.StatusLabel.Text, 'Diagnosing')
                if row > 0, app.StatusLabel.Text = sprintf('Record %d %s diagnosed.', row, char(rid));
                else, app.StatusLabel.Text = 'The typed gas values were diagnosed.';
                end
            end
            ok = true;
        end

        % Sampling times of a run with n oil samples, in hours from the start of the shown 24 h (n = 4 is the Step 40 run) (RD)
        function th = sampleTimesH(~, n)
            if n == 4, th = [6 12 18 22]; else, th = (1:n) * 24 / (n + 1); end
        end

        % Play the sampling events of the last simulation back in time order and diagnose the record of each (D38, RD)
        function playEvents(app, baseText)
            ev = app.SimEvents;  r = app.SimResult;
            app.EventResults = struct('event', {}, 'time_h', {}, 'record_id', {}, 'row', {}, 'hot_spot_C', {}, 'result', {});
            vn = {'Event', 'Time (h)', 'Record', 'Known label', 'Rogers', 'IEC 60599', 'Duval', 'ML', 'Agree (of 3)', 'Verdict', 'Hot-spot (degC)'};
            tbl = table(zeros(0, 1), zeros(0, 1), strings(0, 1), strings(0, 1), strings(0, 1), strings(0, 1), strings(0, 1), strings(0, 1), strings(0, 1), strings(0, 1), zeros(0, 1), 'VariableNames', vn);
            app.EventTable.Data = tbl;
            if isempty(ev)
                app.StatusLabel.Text = [baseText ' No sampling events in this run.'];
                return
            end
            nrec = height(readtable(app.DGAFile));
            app.RecordSpinner.Limits = [1 nrec];
            nok = 0;
            for k = 1:height(ev)
                idx = ev.record_index(k);
                if idx > numel(app.Chosen.rows), continue, end
                row = app.Chosen.rows(idx);  rid = app.Chosen.record_id(idx);
                if isnan(row)
                    app.StatusLabel.Text = sprintf('Event %d: record %s is not in the DGA file.', k, char(rid));
                    continue
                end
                if rid == "", rid = "row " + row; end
                app.RecordSpinner.Value = row;
                app.StatusLabel.Text = sprintf('Sampling event %d at %.1f h: diagnosing %s...', k, ev.time_h(k), char(rid)); drawnow;
                if ~app.diagnoseRow(row), continue, end
                R = app.LastResult;  hs = interp1(r.t_h, r.theta_hs_C, ev.time_h(k));
                mlText = R.ml.status;  agr = "-";
                if R.ml.status == "OK", mlText = R.ml.fault_class; agr = string(R.agreement.count); end
                V = R.verdict;
                if V.health == "fault", vt = "FAULT " + V.fault_class;
                elseif V.health == "healthy", vt = "HEALTHY";
                elseif V.health == "watch", vt = "WATCH";
                else, vt = "CHECK";
                end
                lab = app.Chosen.label(idx);  if lab == "", lab = "-"; end
                tbl(end + 1, :) = {k, ev.time_h(k), rid, lab, string(R.conventional.rogers), string(R.conventional.iec), string(R.conventional.duval), mlText, agr, vt, hs}; %#ok<AGROW>
                app.EventTable.Data = tbl;
                app.EventResults(end + 1) = struct('event', k, 'time_h', ev.time_h(k), 'record_id', rid, 'row', row, 'hot_spot_C', hs, 'result', R);
                nok = nok + 1;
                pause(app.PlaybackPause);
            end
            app.StatusLabel.Text = sprintf('%s %d of %d sampling events diagnosed.', baseText, nok, height(ev));
            if ~isempty(app.EventResults)       % how many of each state, so that the mix of the run is visible at once
                hl = arrayfun(@(e) string(e.result.verdict.health), app.EventResults);
                app.StatusLabel.Text = [app.StatusLabel.Text sprintf(' Verdicts: %d healthy, %d fault, %d watch/check.', sum(hl == "healthy"), sum(hl == "fault"), sum(hl == "watch" | hl == "undecided"))];
            end
            app.ExplainState.events = dashboard_explain('events', app.EventResults);   % summary of the run first, then the last record (MC)
            app.ExplainState.focus = "run";
            app.refreshExplanation();
        end
    end

%%@ CALLBACK RunSimulationButtonPushed Button pushed function: RunSimulationButton
app.setBusy(true);  cleanBusy = onCleanup(@() app.setBusy(false));       % buttons off until the end of the run (MC)
